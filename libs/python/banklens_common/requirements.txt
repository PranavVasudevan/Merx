# Pinned + hashed for reproducible builds (F14). Mirrors pyproject.toml's
# `dependencies` at exact versions, with hashes so `pip install --require-hashes
# -r requirements.txt` rejects anything but the byte-identical wheel. CI uses
# this file directly, then installs banklens_common itself with `--no-deps`
# (its dependencies are already satisfied by this install).
#
# Regenerate after bumping a pin in pyproject.toml:
#   pip download --no-deps -d /tmp/whl "psycopg[binary]==<v>" "duckdb==<v>" \
#     --platform manylinux_2_28_x86_64 --python-version 3.13 \
#     --implementation cp --abi cp313 --only-binary=:all:
#   pip hash /tmp/whl/*.whl
# psycopg-binary is psycopg[binary]'s only real dependency on Linux/py3.13
# (tzdata is win32-only, typing-extensions is python_version<3.13-only — see
# psycopg's own METADATA `Requires-Dist` markers).
